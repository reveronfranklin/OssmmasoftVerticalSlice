using System.Text;
using System.Text.Json;
using Microsoft.Extensions.Configuration;
using Oracle.ManagedDataAccess.Client;
var cfg=new ConfigurationBuilder().AddJsonFile("/srv/ossmmasoft-pruebas/shared/config/vertical.json",false,false).Build();var checks=new List<string>();
void Check(bool v,string n){if(!v)throw new InvalidOperationException(n);checks.Add(n);}
List<string> Blocks(string path){var result=new List<string>();var block=new StringBuilder();foreach(var line in File.ReadLines(path)){if(line.TrimStart().StartsWith("--"))continue;if(line.Trim()=="/"){if(block.ToString().Trim().Length>0)result.Add(block.ToString().Trim());block.Clear();}else block.AppendLine(line);}Check(string.IsNullOrWhiteSpace(block.ToString()),"script-complete");return result;}
string Normalize(string text)=>string.Join("\n",text.Replace("\r","").Split('\n').Select(l=>l.TrimEnd())).Trim();
try{
 using var cn=new OracleConnection(cfg["ConnectionStrings:DefaultConnectionRH"]);await cn.OpenAsync();OracleCommand Cmd(string sql){var c=cn.CreateCommand();c.BindByName=true;c.CommandTimeout=25;c.CommandText=sql;return c;}
 using(var c=Cmd("SELECT SYS_CONTEXT('USERENV','DB_NAME'),SYS_CONTEXT('USERENV','SERVER_HOST'),SYS_CONTEXT('USERENV','SESSION_USER') FROM DUAL"))using(var r=await c.ExecuteReaderAsync()){await r.ReadAsync();Check(cfg.GetValue<int>("settings:EmpresaConfig")==13&&r.GetString(0)==cfg["OracleAnalysis:Targets:RH:ExpectedDatabaseName"]&&r.GetString(1)==cfg["OracleAnalysis:Targets:RH:ExpectedServerHost"]&&r.GetString(2)=="RH","qa-identity");}
 var expected=Blocks(args[1]);Check(expected.Count==2,"expected-feature-package-two-blocks");
 for(var n=0;n<2;n++){
  using var c=Cmd("SELECT TEXT FROM USER_SOURCE WHERE NAME='RH_PKG_NOM_CFG' AND TYPE='"+(n==0?"PACKAGE":"PACKAGE BODY")+"' ORDER BY LINE");var text=new StringBuilder();using var r=await c.ExecuteReaderAsync();while(await r.ReadAsync())text.Append(r.GetString(0));
  Check(Normalize(text.ToString())==Normalize(expected[n].Replace("CREATE OR REPLACE ","")),"feature-package-matches-known-previous-"+n);
 }
 foreach(var sql in Blocks(args[0])){using var c=Cmd(sql);await c.ExecuteNonQueryAsync();}
 using(var c=Cmd("SELECT NAME,LINE,POSITION,TEXT FROM USER_ERRORS WHERE NAME='RH_PKG_NOM_CFG' ORDER BY TYPE,SEQUENCE"))using(var r=await c.ExecuteReaderAsync())while(await r.ReadAsync())Console.WriteLine("R57_SQL_ERROR="+r.GetString(0)+":"+r.GetValue(1)+":"+r.GetValue(2)+" "+r.GetString(3));
 using(var c=Cmd("SELECT COUNT(*) FROM USER_OBJECTS WHERE OBJECT_NAME='RH_PKG_NOM_CFG' AND OBJECT_TYPE IN ('PACKAGE','PACKAGE BODY') AND STATUS='VALID'"))Check(Convert.ToInt32(await c.ExecuteScalarAsync())==2,"feature-package-valid");
 using(var c=Cmd("SELECT COUNT(*) FROM USER_CONSTRAINTS WHERE TABLE_NAME='RH_NOM_VAR' AND CONSTRAINT_NAME='CK_RH_NOM_VAR_TIP2' AND STATUS='ENABLED' AND VALIDATED='VALIDATED'"))Check(Convert.ToInt32(await c.ExecuteScalarAsync())==1,"typed-constraint-validated");
 using(var c=Cmd("SELECT COUNT(*) FROM USER_CONSTRAINTS WHERE TABLE_NAME='RH_NOM_VAR' AND CONSTRAINT_NAME IN ('CK_RH_NOM_VAR_TIPO','CK_RH_NOM_VAR_UNI')"))Check(Convert.ToInt32(await c.ExecuteScalarAsync())==0,"old-numeric-only-constraints-replaced");
 Console.WriteLine("R57_INSTALL="+JsonSerializer.Serialize(new{Status="OK",Checks=checks,NoPayrollWrites=true,LegacyPackageChanged=false}));
}catch(OracleException ex){Console.WriteLine("R57_FAIL_ORACLE="+ex.Number);Environment.ExitCode=1;}catch(InvalidOperationException ex){Console.WriteLine("R57_FAIL_CHECK="+ex.Message);Environment.ExitCode=1;}
