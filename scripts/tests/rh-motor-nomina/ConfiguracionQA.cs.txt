async Task<OracleCommand> Query(OracleConnection cn,string sql,params (string,object)[] binds){var cmd=cn.CreateCommand();cmd.CommandText=sql;cmd.BindByName=true;cmd.CommandTimeout=20;foreach(var(n,v) in binds)cmd.Parameters.Add(n,v is int?OracleDbType.Int32:OracleDbType.Varchar2).Value=v;await Task.CompletedTask;return cmd;}
async Task<int> Scalar(OracleConnection cn,string sql,params (string,object)[] binds){using var q=await Query(cn,sql,binds);return Convert.ToInt32(await q.ExecuteScalarAsync());}
async Task Identity(OracleConnection cn,string alias){await cn.OpenAsync();using var q=await Query(cn,"SELECT SYS_CONTEXT('USERENV','DB_NAME'),SYS_CONTEXT('USERENV','SERVER_HOST'),SYS_CONTEXT('USERENV','SESSION_USER') FROM DUAL");using var r=await q.ExecuteReaderAsync();await r.ReadAsync();var expected=config.GetSection("OracleAnalysis:Targets:"+alias);Check(r.GetString(0)==expected["ExpectedDatabaseName"] && r.GetString(1)==expected["ExpectedServerHost"] && r.GetString(2)==alias,"identity-"+alias);}
async Task<string> OldHash(){using var q=await Query(rh,"SELECT TEXT FROM USER_SOURCE WHERE NAME='RH_PKG_ADJUNTAR_CARGO' ORDER BY TYPE,LINE");using var r=await q.ExecuteReaderAsync();var b=new StringBuilder();while(await r.ReadAsync())b.Append(r.IsDBNull(0)?"":r.GetString(0));return Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(b.ToString())));}

 foreach(var file in new[]{"00_SIS_NOM_CFG_GRANT.sql","01_RH_NOM_CFG_TABLES.sql","02_RH_PKG_NOM_CFG.sql"}){
  var raw=await File.ReadAllTextAsync(Path.Combine(scripts,file));
  var clean=string.Join("\n",raw.Split('\n').Where(l=>!l.TrimStart().StartsWith("--")));
  foreach(var block in Regex.Split(clean,@"(?m)^\s*/\s*$")){
   var sql=block.Trim();if(sql.Length==0)continue;if(sql.StartsWith("GRANT",StringComparison.OrdinalIgnoreCase))sql=sql.TrimEnd(';');
   using var cmd=await Query(file.StartsWith("00_")?sis:rh,sql);await cmd.ExecuteNonQueryAsync();
  }
  Check(true,"installed-"+file);
 }
 var errors=new List<string>();
 using(var q=await Query(rh,"SELECT NAME||'/'||TYPE||' line '||TO_CHAR(LINE)||': '||TEXT FROM USER_ERRORS WHERE NAME IN ('RH_PKG_NOM_CFG','TR_RH_NOM_VER_IMM') ORDER BY NAME,TYPE,SEQUENCE"))
 using(var r=await q.ExecuteReaderAsync())while(await r.ReadAsync())errors.Add(r.GetString(0));
 if(errors.Count>0){File.WriteAllLines(Path.Combine(results,"compile-errors.txt"),errors);Console.WriteLine(string.Join("\n",errors));throw new InvalidOperationException("Objects have compile errors");}
 Check(await Scalar(rh,"SELECT COUNT(*) FROM USER_OBJECTS WHERE OBJECT_NAME IN ('RH_PKG_NOM_CFG','TR_RH_NOM_VER_IMM') AND STATUS<>'VALID'")==0,"oracle-objects-valid");
 Check(await OldHash()==oldHash,"old-payroll-package-unchanged");
 phase="fixtures";
 var tag="Q57_"+Guid.NewGuid().ToString("N")[..8].ToUpperInvariant();
 var actor=int.MaxValue;
 while(await Scalar(sis,"SELECT COUNT(*) FROM SIS.SIS_USUARIOS WHERE CODIGO_USUARIO=:id",("id",actor))>0)actor--;
 var model=0;
 using(var tx=sis.BeginTransaction()){
  using(var q=await Query(sis,"LOCK TABLE SIS.OSS_MODELO_CALCULO IN SHARE ROW EXCLUSIVE MODE NOWAIT")){q.Transaction=tx;await q.ExecuteNonQueryAsync();}
  var occupied=new HashSet<int>();
  using(var q=await Query(sis,"SELECT ID FROM SIS.OSS_MODELO_CALCULO WHERE ID BETWEEN 1 AND 10000")){q.Transaction=tx;using var r=await q.ExecuteReaderAsync();while(await r.ReadAsync())occupied.Add(r.GetInt32(0));}
  // Avoid attaching the fixture to concepts with orphaned model references.
  using(var q=await Query(rh,"SELECT DISTINCT ID_MODELO_CALCULO FROM RH.RH_CONCEPTOS WHERE CODIGO_EMPRESA=:company AND ID_MODELO_CALCULO BETWEEN 1 AND 10000",("company",empresa)))
  {using var r=await q.ExecuteReaderAsync();while(await r.ReadAsync())occupied.Add(r.GetInt32(0));}
  model=Enumerable.Range(1,10000).First(i=>!occupied.Contains(i)); // Bounded gap allocator under lock, not MAX+1.
  using(var q=await Query(sis,"INSERT INTO SIS.OSS_MODELO_CALCULO(ID,DESCRIPCION,CODIGO_EMPRESA,USUARIO_INS,FECHA_INS) VALUES(:id,:description,:company,:actor,SYSDATE)",("id",model),("description",tag+" QA TEST NO USAR"),("company",empresa),("actor",actor))){q.Transaction=tx;await q.ExecuteNonQueryAsync();}
  await tx.CommitAsync();
 }

 var vars=new RhNomVariablesHandler(db);var models=new RhNomVersionesHandler(db);
